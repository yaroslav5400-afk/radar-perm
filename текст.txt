        // Render Clean Regions
        function renderRegions() {
            regionData.forEach(reg => {
                const geoJson = {
                    "type": "Feature",
                    "properties": { "name": reg.name, "status": reg.status, "lastUpdate": reg.lastUpdate },
                    "geometry": { "type": "Polygon", "coordinates": reg.coords }
                };

                const strokeColor = reg.status === 'danger' ? '#ef4444' : reg.status === 'warning' ? '#f59e0b' : '#10b981';

                const layer = L.geoJSON(geoJson, {
                    style: {
                        color: strokeColor,
                        fillColor: reg.color,
                        fillOpacity: 0.2,
                        weight: 2,
                        dashArray: '3, 6'
                    },
                    onEachFeature: (feature, l) => {
                        l.on('mouseover', function() {
                            this.setStyle({ fillOpacity: 0.45, weight: 3 });
                        });
                        l.on('mouseout', function() {
                            this.setStyle({ fillOpacity: 0.2, weight: 2 });
                        });
                        l.on('click', function(e) {
                            L.DomEvent.stopPropagation(e);
                            map.flyTo(reg.center, reg.zoom, { duration: 1.2 });
                        });

                        const statusText = reg.status === 'danger' ? 'ОПАСНОСТЬ БПЛА / АТАКА' : reg.status === 'warning' ? 'ПОВЫШЕННАЯ ГОТОВНОСТЬ' : 'ОБСТАНОВКА СПОКОЙНАЯ';
                        const statusClass = reg.status === 'danger' ? 'text-red-400' : reg.status === 'warning' ? 'text-amber-400' : 'text-emerald-400';

                        l.bindPopup(`
                            <div class="p-2.5 text-xs space-y-1">
                                <div class="font-bold text-sm text-white border-b border-slate-700/80 pb-1">${reg.name}</div>
                                <div class="text-slate-300 mt-1">Статус: <span class="font-bold ${statusClass}">${statusText}</span></div>
                                <div class="text-[10px] text-slate-400">Обновлено: ${reg.lastUpdate}</div>
                            </div>
                        `);
                    }
                }).addTo(map);

                regionLayers.push(layer);
            });

                const marker = L.marker([obj.lat, obj.lng], { icon: customIcon }).addTo(map);
                marker.bindPopup(`
                    <div class="p-2 text-xs space-y-1">
                        <div class="font-bold text-white border-b border-slate-700 pb-1 flex justify-between items-center">
                            <span>Объект ${obj.id}</span>
                            <span class="text-[10px] px-1.5 py-0.5 rounded ${obj.status === 'danger' ? 'bg-red-900/80 text-red-200' : 'bg-emerald-900/80 text-emerald-200'}">${obj.type}</span>
                        </div>
                        <div class="text-slate-300">Скорость: <span class="font-mono text-sky-300">${obj.speed}</span></div>
                        <div class="text-slate-300">Высота: <span class="font-mono text-sky-300">${obj.alt}</span></div>
                        <div class="text-slate-300">Курс: <span class="font-mono text-sky-300">${obj.heading}°</span></div>
                    </div>
                `);

                marker.objData = obj;
                uavMarkers.push(marker);
            });
        }

        // Render Airport Markers
        function renderAirports() {
            airportMarkers.forEach(m => map.removeLayer(m));
            airportMarkers = [];

            if (!showAirports) return;

            airportObjects.forEach(ap => {
                const iconHtml = `
                    <div class="w-7 h-7 rounded-lg bg-slate-900/90 border border-sky-400/80 text-sky-400 flex items-center justify-center shadow-lg">
                        <i class="fa-solid fa-plane-departure text-xs"></i>
                    </div>
                `;
                const icon = L.divIcon({
                    html: iconHtml,
                    className: 'ap-marker',
                    iconSize: [28, 28],
                    iconAnchor: [14, 14]
                });

                const marker = L.marker([ap.lat, ap.lng], { icon: icon }).addTo(map);
                marker.bindPopup(`
                    <div class="p-2 text-xs">
                        <div class="font-bold text-sky-300">${ap.name}</div>
                        <div class="text-slate-400">${ap.city}</div>
                        <div class="mt-1 text-slate-300">Режим: <span class="text-emerald-400 font-semibold">${ap.status}</span></div>
                    </div>
                `);
                airportMarkers.push(marker);
            });
        }

        // Populate Dropdown
        function populateRegionDropdown() {
            const select = document.getElementById('region-dropdown');
            regionData.forEach(reg => {
                const opt = document.createElement('option');
                opt.value = reg.id;
                opt.textContent = reg.name;
                select.appendChild(opt);
            });

            select.addEventListener('change', (e) => {
                const id = e.target.value;
                const reg = regionData.find(r => r.id === id);
                if (reg) {
                    map.flyTo(reg.center, reg.zoom, { duration: 1.2 });
                }
            });
        }

        // Toggle Layers
        function toggleLayer(type) {
            if (type === 'uav') {
                showUAV = !showUAV;
                document.getElementById('btn-uav').classList.toggle('text-emerald-400');
                document.getElementById('btn-uav').classList.toggle('text-slate-500');
                renderUAVs();
            } else if (type === 'airports') {
                showAirports = !showAirports;
                document.getElementById('btn-airports').classList.toggle('text-sky-400');
                document.getElementById('btn-airports').classList.toggle('text-slate-500');
                renderAirports();
            }
        }

        // Live Simulation Movement
        function toggleSimulation() {
            isSimulating = !isSimulating;
            const icon = document.getElementById('sim-icon');

            if (isSimulating) {
                icon.className = "fa-solid fa-pause text-amber-400";
                simInterval = setInterval(() => {
                    uavMarkers.forEach(marker => {
                        const obj = marker.objData;
                        const rad = (obj.heading * Math.PI) / 180;
                        const step = 0.015;

                        obj.lat += Math.cos(rad) * step;
                        obj.lng += Math.sin(rad) * step;

                        marker.setLatLng([obj.lat, obj.lng]);
                    });
                }, 800);
            } else {
                icon.className = "fa-solid fa-play text-amber-400";
                clearInterval(simInterval);
            }
        }

        // Ruler Tool Implementation
        function toggleRuler() {
            isRulerActive = !isRulerActive;
            const banner = document.getElementById('ruler-banner');
            banner.classList.toggle('hidden', !isRulerActive);
            if (!isRulerActive) clearRuler();
        }

        function onMapClick(e) {
            if (!isRulerActive) return;

            rulerPoints.push(e.latlng);

            const circle = L.circleMarker(e.latlng, {
                radius: 4,
                color: '#0ea5e9',
                fillColor: '#0ea5e9',
                fillOpacity: 1
            }).addTo(map);
            rulerMarkers.push(circle);

            if (rulerPoints.length > 1) {
                if (rulerPolyline) {
                    rulerPolyline.setLatLngs(rulerPoints);
                } else {
                    rulerPolyline = L.polyline(rulerPoints, { color: '#0ea5e9', weight: 3, dashArray: '5, 8' }).addTo(map);
                }

                let dist = 0;
                for (let i = 0; i < rulerPoints.length - 1; i++) {
                    dist += rulerPoints[i].distanceTo(rulerPoints[i+1]);
                }
                document.getElementById('ruler-distance').innerText = `${(dist / 1000).toFixed(1)} км`;
            }
        }

        function clearRuler() {
            rulerPoints = [];
            if (rulerPolyline) { map.removeLayer(rulerPolyline); rulerPolyline = null; }
            rulerMarkers.forEach(m => map.removeLayer(m));
            rulerMarkers = [];
            document.getElementById('ruler-distance').innerText = '0 км';
        }

            document.getElementById('toast-title').innerText = title;
            document.getElementById('toast-desc').innerText = desc;
            document.getElementById('toast-banner').classList.remove('hidden');
        }

        function closeToast() {
